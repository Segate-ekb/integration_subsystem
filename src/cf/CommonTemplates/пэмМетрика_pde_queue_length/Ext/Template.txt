// Метрика: Длина очередей сообщений (pde_queue_length)
// Лейблы: queue (incoming/outgoing/distribution)
//
// Единица измерения - ожидающая выполнения операция: во входящей очереди это обработка
// сообщения, в исходящей - формирование и отправка сообщения, в очереди рассылки - отправка
// сообщения одному подписчику. Сообщение, ожидающее рассылки пяти подписчикам, вносит
// в значение метки distribution пять единиц.
//
// Каждая ветвь объединения возвращает ровно одну строку итога, поэтому все три временные
// последовательности присутствуют в ответе, а пустая очередь выводится с нулём.

ТаблицаЗначений = Новый ТаблицаЗначений;
ТаблицаЗначений.Колонки.Добавить("queue", Новый ОписаниеТипов("Строка"));
ТаблицаЗначений.Колонки.Добавить("value", Новый ОписаниеТипов("Число"));

Условия = пэмМетрикиСервер.УсловияОтбораОперацийОчередей();

Запрос = Новый Запрос;
Запрос.Текст = СтрШаблон(
"ВЫБРАТЬ
|	""incoming"" КАК queue,
|	КОЛИЧЕСТВО(*) КАК value
|ИЗ
|	РегистрСведений.инт_ТекущийСтатусВходящихСообщений КАК Статус
|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений.инт_ОчередьВходящихСообщений КАК Очередь
|		ПО Статус.ИдентификаторСообщения = Очередь.ИдентификаторСообщения
|		ЛЕВОЕ СОЕДИНЕНИЕ Справочник.инт_ПотокиДанных КАК Поток
|		ПО Очередь.ПотокДанных = Поток.Ссылка
|ГДЕ
|	%1
|
|ОБЪЕДИНИТЬ ВСЕ
|
|ВЫБРАТЬ
|	""outgoing"",
|	КОЛИЧЕСТВО(*)
|ИЗ
|	РегистрСведений.инт_ТекущийСтатусИсходящихСообщений КАК Статус
|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений.инт_ОчередьИсходящихСообщений КАК Очередь
|		ПО Статус.ИдентификаторСообщения = Очередь.ИдентификаторСообщения
|		ЛЕВОЕ СОЕДИНЕНИЕ Справочник.инт_ПотокиДанных КАК Поток
|		ПО Очередь.ПотокДанных = Поток.Ссылка
|ГДЕ
|	%2
|
|ОБЪЕДИНИТЬ ВСЕ
|
|ВЫБРАТЬ
|	""distribution"",
|	КОЛИЧЕСТВО(*)
|ИЗ
|	РегистрСведений.инт_ТекущийСтатусРассылкиСообщений КАК Статус
|ГДЕ
|	%3",
	Условия.Входящие.Ожидающие, Условия.Исходящие.Ожидающие, Условия.Рассылка.Ожидающие);

Результат = Запрос.Выполнить();
Если НЕ Результат.Пустой() Тогда
	ТаблицаЗначений = Результат.Выгрузить();
КонецЕсли;
