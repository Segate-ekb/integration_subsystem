// Метрика: Возраст старейшей ожидающей выполнения операции (pde_oldest_pending_age_seconds)
// Лейблы: queue (incoming/outgoing/distribution)
//
// В расчёт входят те же ожидающие выполнения операции, что считает pde_queue_length:
// операция, которую подсистема больше не возьмёт в работу, возраст очереди не увеличивает.
// Единицей работы в очереди рассылки служит пара «сообщение, подписчик», поэтому история
// статусов соединяется по обоим измерениям регистра.
//
// Возраст отсчитывается от наиболее ранней отметки истории статусов операции. Операция,
// у которой отметок истории нет, из отбора не исключается, но возраста не имеет и итог
// не занижает: минимум берётся по остальным операциям очереди.

ТаблицаЗначений = Новый ТаблицаЗначений;
ТаблицаЗначений.Колонки.Добавить("queue", Новый ОписаниеТипов("Строка"));
ТаблицаЗначений.Колонки.Добавить("value", Новый ОписаниеТипов("Число"));

Условия = пэмМетрикиСервер.УсловияОтбораОперацийОчередей();

Запрос = Новый Запрос;
Запрос.Текст = СтрШаблон(
"ВЫБРАТЬ
|	""incoming"" КАК queue,
|	ЕСТЬNULL(РАЗНОСТЬДАТ(МИНИМУМ(История.Период), &МоментРасчета, СЕКУНДА), 0) КАК value
|ИЗ
|	РегистрСведений.инт_ТекущийСтатусВходящихСообщений КАК Статус
|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений.инт_ОчередьВходящихСообщений КАК Очередь
|		ПО Статус.ИдентификаторСообщения = Очередь.ИдентификаторСообщения
|		ЛЕВОЕ СОЕДИНЕНИЕ Справочник.инт_ПотокиДанных КАК Поток
|		ПО Очередь.ПотокДанных = Поток.Ссылка
|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений.инт_ИсторияСтатусовВходящихСообщений КАК История
|		ПО Статус.ИдентификаторСообщения = История.ИдентификаторСообщения
|ГДЕ
|	%1
|
|ОБЪЕДИНИТЬ ВСЕ
|
|ВЫБРАТЬ
|	""outgoing"",
|	ЕСТЬNULL(РАЗНОСТЬДАТ(МИНИМУМ(История.Период), &МоментРасчета, СЕКУНДА), 0)
|ИЗ
|	РегистрСведений.инт_ТекущийСтатусИсходящихСообщений КАК Статус
|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений.инт_ОчередьИсходящихСообщений КАК Очередь
|		ПО Статус.ИдентификаторСообщения = Очередь.ИдентификаторСообщения
|		ЛЕВОЕ СОЕДИНЕНИЕ Справочник.инт_ПотокиДанных КАК Поток
|		ПО Очередь.ПотокДанных = Поток.Ссылка
|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений.инт_ИсторияСтатусовИсходящихСообщений КАК История
|		ПО Статус.ИдентификаторСообщения = История.ИдентификаторСообщения
|ГДЕ
|	%2
|
|ОБЪЕДИНИТЬ ВСЕ
|
|ВЫБРАТЬ
|	""distribution"",
|	ЕСТЬNULL(РАЗНОСТЬДАТ(МИНИМУМ(История.Период), &МоментРасчета, СЕКУНДА), 0)
|ИЗ
|	РегистрСведений.инт_ТекущийСтатусРассылкиСообщений КАК Статус
|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений.инт_ИсторияСтатусовРассылкиСообщений КАК История
|		ПО Статус.ИдентификаторСообщения = История.ИдентификаторСообщения
|			И Статус.Подписчик = История.Подписчик
|ГДЕ
|	%3",
	Условия.Входящие.Ожидающие, Условия.Исходящие.Ожидающие, Условия.Рассылка.Ожидающие);

// История статусов записывается по времени сеанса, поэтому и момент расчёта берётся из него.
Запрос.УстановитьПараметр("МоментРасчета", ТекущаяДатаСеанса());

Результат = Запрос.Выполнить();
Если НЕ Результат.Пустой() Тогда
	ТаблицаЗначений = Результат.Выгрузить();
КонецЕсли;
