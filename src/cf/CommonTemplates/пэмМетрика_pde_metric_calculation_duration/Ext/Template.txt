// Метрика: Длительность расчёта метрики (pde_metric_calculation_duration_seconds)
// Лейблы: metric - код метрики справочника пэмМетрики
//
// Значение - длительность последнего расчёта метрики в секундах. Регистр хранит длительность
// в целых миллисекундах, поэтому разрешение измерения равно одной миллисекунде, а расчёт
// быстрее миллисекунды выгружается нулём.
//
// Состояние расчёта записывается после формирования строки ответа, поэтому значения метрики
// относятся к предыдущему циклу расчёта.
//
// Деление выполняется после выборки: колонка результата запроса унаследовала бы от ресурса
// регистра целочисленный тип и дробная часть была бы потеряна.

МиллисекундВСекунде = 1000;

ТаблицаЗначений = Новый ТаблицаЗначений;
ТаблицаЗначений.Колонки.Добавить("metric", Новый ОписаниеТипов("Строка"));
ТаблицаЗначений.Колонки.Добавить("value", Новый ОписаниеТипов("Число"));

Запрос = Новый Запрос;
Запрос.Текст =
"ВЫБРАТЬ
|	Метрики.Код КАК metric,
|	СостояниеМетрик.Длительность КАК Длительность
|ИЗ
|	РегистрСведений.пэмСостояниеМетрик КАК СостояниеМетрик
|		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Справочник.пэмМетрики КАК Метрики
|		ПО СостояниеМетрик.Метрика = Метрики.Ссылка
|ГДЕ
|	Метрики.Активность";

Выборка = Запрос.Выполнить().Выбрать();
Пока Выборка.Следующий() Цикл
	НоваяСтрока = ТаблицаЗначений.Добавить();
	НоваяСтрока.metric = Выборка.metric;
	НоваяСтрока.value = Выборка.Длительность / МиллисекундВСекунде;
КонецЦикла;
