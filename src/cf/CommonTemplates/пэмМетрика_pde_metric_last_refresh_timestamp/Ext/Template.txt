// Метрика: Момент последнего расчёта метрики (pde_metric_last_refresh_timestamp_seconds)
// Лейблы: metric - код метрики справочника пэмМетрики
//
// Значение - момент завершения последнего успешного расчёта метрики в секундах эпохи Unix.
// Регистр хранит момент расчёта во времени базы, поэтому на границе с Prometheus он приводится
// к универсальному времени: секунды эпохи Unix универсальны по определению, и возраст расчёта
// потребитель вычисляет выражением time() - pde_metric_last_refresh_timestamp_seconds.
//
// Пока ни одна метрика не рассчитана, регистр пуст и метрика в ответ не попадает: отметка
// времени несостоявшегося расчёта не определена.

ТаблицаЗначений = Новый ТаблицаЗначений;
ТаблицаЗначений.Колонки.Добавить("metric", Новый ОписаниеТипов("Строка"));
ТаблицаЗначений.Колонки.Добавить("value", Новый ОписаниеТипов("Число"));

Запрос = Новый Запрос;
Запрос.Текст =
"ВЫБРАТЬ
|	Метрики.Код КАК metric,
|	СостояниеМетрик.ДатаРасчета КАК ДатаРасчета
|ИЗ
|	РегистрСведений.пэмСостояниеМетрик КАК СостояниеМетрик
|		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Справочник.пэмМетрики КАК Метрики
|		ПО СостояниеМетрик.Метрика = Метрики.Ссылка
|ГДЕ
|	Метрики.Активность";

НачалоЭпохи = Дата(1970, 1, 1);
ЧасовойПоясХранения = инт_ОбщегоНазначения.ЧасовойПоясВремениБазы();

Выборка = Запрос.Выполнить().Выбрать();
Пока Выборка.Следующий() Цикл
	СтрокаМетрики = ТаблицаЗначений.Добавить();
	СтрокаМетрики.metric = Выборка.metric;
	СтрокаМетрики.value = УниверсальноеВремя(Выборка.ДатаРасчета, ЧасовойПоясХранения) - НачалоЭпохи;
КонецЦикла;
